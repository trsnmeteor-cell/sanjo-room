未設定（この端末のみで動作）

設定手順（初回のみ）
  1. Googleドライブでスプレッドシートを新規作成（名前は任意）
  2. メニューの拡張機能 → Apps Script を開く
  3. 下のコードを貼り付けて保存（既存の内容は消してOK）
  4. デプロイ → 新しいデプロイ → 種類の選択で「ウェブアプリ」
  5. 実行ユーザー＝自分／アクセスできるユーザー＝全員 にして「デプロイ」
  6. 表示されたウェブアプリのURLを上欄に貼り付け →「接続テスト」→「保存して同期」

基準（正）はスプレッドシートの room_info です。送るのは変更した列だけ（機種／モニター／エアコン／カメラ／清掃日／清掃者／作業完了／異常）で、他の列と写真はそのまま残ります。送る前に「自分が前回知っていた値」と突き合わせ、サーバー側が先に変わっていれば書かずに競合として記録します（内容が黙って消えることはありません）。

2台目以降の端末も、同じURLを貼るだけで同じ台帳を共有できます。すでに設定済みの場合は、下のコードで「デプロイを管理 → 編集 → 新しいバージョン」として再デプロイしてください（同時登録の取りこぼし防止・写真を同期から外して軽量化、が入っています）。
写真は同期時に転送せず、部屋を開いたときだけ取得します。台帳の内容（機種・型番・日付など）は数十KBなので、写真が重くても同期は止まりません。

/* ============================================================
   三条河原町 部屋台帳 — 共有サーバー（Google Apps Script）
   v8：シートは room_info の1枚だけを使います。
       ・room_state（旧・key-value形式）は使いません
       ・conflicts / room_conflicts（競合記録）はシートに作りません
       ・staff（担当者リスト）はシートに作りません
   ============================================================ */
const SHEET_NAME = 'room_info';
const CODE_VERSION = 11;

/* room_info の列（1始まり）。ここが「正」の情報。 */
const COLS = {
  room: 1, floor: 2, model: 3, monitor: 4, ac: 5, camera: 6,
  cleanedOn: 7, cleaner: 8, work: 9, incident: 10,
  updatedAt: 11, author: 12, photo: 13, seq: 14
};
const HEADERS = ['部屋番号', '階', '機種', 'モニター型番', 'エアコン型番', 'カメラ',
  '清掃実施日', '清掃実施者', '作業完了', '異常', '更新日時', '更新端末', '写真', '版'];
/* 競合判定は写真を除く（写真は大きすぎて突き合わせが脆く、行ごと拒否されてしまうため） */
const INFO_KEYS = ['model', 'monitor', 'ac', 'camera', 'cleanedOn', 'cleaner', 'work', 'incident'];
/* 日付の列は「スプレッドシートが日付セルに自動変換する」ため、読み書きと比較で表記を揃える。
   揃えないと、同じ日でも「2026-10-15」と「Thu Oct 15 2026 ...」が食い違い、行ごと拒否される。 */
const DATE_COLS = [7, 11];
const BIG_PHOTO = 45000;
/* もう使わないシート（設定の「古いシートを削除する」で消せます） */
const LEGACY_SHEETS = ['room_state', 'conflicts', 'room_conflicts', 'staff'];

/* 読み取り（GET）。?cb= があれば JSONP。 */
function doGet(e) {
  const p = (e && e.parameter) || {};
  const cb = p.cb || '';
  let payload;
  try {
    if (p.action === 'ping') {
      payload = { ok: true, pong: true, version: CODE_VERSION, serverTime: new Date().toISOString(), rows: [] };
    } else if (p.action === 'init') {
      payload = initTable_(p.rooms || '');
    } else if (p.action === 'repair') {
      payload = repairTable_();
    } else if (p.action === 'cleanup') {
      payload = cleanupSheets_();
    } else if (p.action === 'upsert') {
      payload = applyRows_(p.rows || '', p.author || '');
    } else {
      payload = list_(p);
    }
  } catch (err) {
    payload = { ok: false, error: String(err) };
  }
  const body = cb ? (cb + '(' + JSON.stringify(payload) + ')') : JSON.stringify(payload);
  return ContentService.createTextOutput(body)
    .setMimeType(cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON);
}

/* POST（互換用）。通常は GET（302転送で本文が失われる環境があるため） */
function doPost(e) {
  let result;
  try {
    const body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    result = (body.action === 'upsert') ? applyRows_(body.rows || [], body.author || '') : list_({});
  } catch (err) {
    result = { ok: false, error: String(err) };
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

/* ============================================================
   部屋一覧を用意する（何度呼んでも重複しない）
   - 足りない部屋だけを1回の書き込みでまとめて追加
   - 既に重複している行は1行に統合（版が大きい行＝新しい行を残す）
   - 完全に空の行は削除
   ============================================================ */
function initTable_(roomsRaw) {
  const sh = sheet_();
  const lock = LockService.getScriptLock();
  let created = 0, total = 0, fixed = { merged: 0, deletedEmpty: 0 };
  try {
    lock.waitLock(20000);
    fixed = prune_(sh);                       /* まず既存の重複・空行を整理 */
    const last = sh.getLastRow();
    const seen = {};
    if (last > 1) {
      const vals = sh.getRange(2, COLS.room, last - 1, 1).getValues();
      vals.forEach(function (r) { const k = String(r[0] || '').trim(); if (k) seen[k] = true; });
    }
    const rows = [];
    String(roomsRaw || '').split(',').forEach(function (pair) {
      pair = String(pair).trim();
      if (!pair) return;
      const parts = pair.split(':');
      const room = String(parts[0] || '').trim();
      const floor = String(parts[1] || '').trim();
      if (!room || seen[room]) return;
      const row = new Array(HEADERS.length).fill('');
      row[COLS.room - 1] = room;
      row[COLS.floor - 1] = floor;
      rows.push(row);
      seen[room] = true;
      created++;
    });
    if (rows.length) {                        /* まとめて1回で書き込む（途中で重なる隙を作らない） */
      sh.getRange(sh.getLastRow() + 1, 1, rows.length, HEADERS.length).setValues(rows);
    }
    const again = prune_(sh);                 /* 同時実行で増えていた場合もここで統合 */
    fixed.merged += again.merged;
    fixed.deletedEmpty += again.deletedEmpty;
    total = sh.getLastRow() - 1;
  } catch (err) {
    return { ok: false, error: String(err) };
  } finally {
    try { lock.releaseLock(); } catch (err) {}
  }
  return {
    ok: true, version: CODE_VERSION, created: created, total: total,
    merged: fixed.merged, deletedEmpty: fixed.deletedEmpty,
    serverTime: new Date().toISOString(), rows: []
  };
}

/* 重複と空行の整理だけを行う */
function repairTable_() {
  const sh = sheet_();
  const lock = LockService.getScriptLock();
  let fixed = { merged: 0, deletedEmpty: 0 }, total = 0;
  try {
    lock.waitLock(20000);
    fixed = prune_(sh);
    total = sh.getLastRow() - 1;
  } catch (err) {
    return { ok: false, error: String(err) };
  } finally {
    try { lock.releaseLock(); } catch (err) {}
  }
  return {
    ok: true, version: CODE_VERSION, merged: fixed.merged, deletedEmpty: fixed.deletedEmpty,
    total: total, serverTime: new Date().toISOString(), rows: []
  };
}

/* 旧シート（room_state / conflicts / room_conflicts / staff）を削除する。room_info は消しません。 */
function cleanupSheets_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const deleted = [], kept = [];
  try {
    LEGACY_SHEETS.forEach(function (name) {
      const sh = ss.getSheetByName(name);
      if (!sh) return;
      if (ss.getSheets().length <= 1) return;      /* 最後の1枚は残す */
      ss.deleteSheet(sh);
      deleted.push(name);
    });
    ss.getSheets().forEach(function (s) { kept.push(s.getName()); });
  } catch (err) {
    return { ok: false, error: String(err) };
  }
  return {
    ok: true, version: CODE_VERSION, deleted: deleted, sheets: kept,
    serverTime: new Date().toISOString(), rows: []
  };
}

/* 行の「濃さ」。版が大きい行を優先し、同じなら記入が多い行を残す */
function score_(r) {
  let filled = 0;
  for (let i = 0; i < r.length; i++) { if (String(r[i] === null || r[i] === undefined ? '' : r[i]).trim() !== '') filled++; }
  return Number(r[COLS.seq - 1] || 0) * 1000 + filled;
}

/* 同じ部屋番号の行を1本にまとめ、空行を消す */
function prune_(sh) {
  const last = sh.getLastRow();
  if (last < 2) return { merged: 0, deletedEmpty: 0 };
  const values = sh.getRange(2, 1, last - 1, HEADERS.length).getValues();
  const best = {};
  const drop = {};
  let merged = 0, deletedEmpty = 0;
  for (let i = 0; i < values.length; i++) {
    const room = String(values[i][COLS.room - 1] || '').trim();
    if (!room) { drop[i] = true; deletedEmpty++; continue; }      /* 部屋番号が空＝消す */
    if (!(room in best)) { best[room] = i; continue; }
    if (score_(values[i]) > score_(values[best[room]])) { drop[best[room]] = true; best[room] = i; }
    else drop[i] = true;
    merged++;
  }
  const idx = Object.keys(drop).map(Number).sort(function (a, b) { return b - a; });
  let run = 0, runStart = -1;
  idx.forEach(function (i) {
    if (run > 0 && i === runStart - 1) { run++; runStart = i; return; }
    if (run > 0) sh.deleteRows(runStart + 2, run);
    run = 1; runStart = i;
  });
  if (run > 0) sh.deleteRows(runStart + 2, run);
  return { merged: merged, deletedEmpty: deletedEmpty };
}

/* 一覧を返す。light=1 で写真を外す（同期用）。keys= で部屋を絞る（写真つき） */
function list_(p) {
  const sh = sheet_();
  const last = sh.getLastRow();
  const want = p.keys ? String(p.keys).split(',').map(function (k) { return String(k).replace(/^room:/, '').trim(); }) : null;
  const light = String(p.light || '') === '1';
  const out = [];
  if (last > 1) {
    const values = sh.getRange(2, 1, last - 1, HEADERS.length).getValues();
    values.forEach(function (r) {
      const room = String(r[COLS.room - 1] || '').trim();
      if (!room) return;
      if (want && want.indexOf(room) < 0) return;
      out.push(rowToObj(r, room, light));
    });
  }
  /* 万一重複が残っていても、同じ部屋は1件だけ返す（版が大きい行を優先） */
  const best = {}, picked = [];
  out.forEach(function (o) {
    if (!(o.room in best)) { best[o.room] = picked.length; picked.push(o); return; }
    if (Number(o.seq || 0) > Number(picked[best[o.room]].seq || 0)) picked[best[o.room]] = o;
  });
  return { ok: true, version: CODE_VERSION, serverTime: new Date().toISOString(), count: picked.length, rows: picked };
}

function dateStr_(v, full) {
  if (v === null || v === undefined || v === '') return '';
  if (v instanceof Date) {
    const tz = Session.getScriptTimeZone() || 'Asia/Tokyo';
    return Utilities.formatDate(v, tz, full ? 'yyyy/MM/dd HH:mm:ss' : 'yyyy-MM-dd');
  }
  return String(v);
}
function canon_(name, v) {
  const t = (v === null || v === undefined) ? '' : String(v).trim();
  if (name === 'cleanedOn') {
    const d = new Date(t);
    if (!isNaN(d.getTime())) {
      const tz = Session.getScriptTimeZone() || 'Asia/Tokyo';
      return Utilities.formatDate(d, tz, 'yyyy-MM-dd');
    }
  }
  return t;
}
function rowToObj(r, room, light) {
  function s(i) { const v = r[i - 1]; return (v === null || v === undefined) ? '' : String(v); }
  function sd(i, full) { const v = r[i - 1]; return (v === null || v === undefined || v === '') ? '' : dateStr_(v, full); }
  const photo = s(COLS.photo);
  const fields = {
    model: s(COLS.model), monitor: s(COLS.monitor), ac: s(COLS.ac), camera: s(COLS.camera),
    cleanedOn: sd(COLS.cleanedOn), cleaner: s(COLS.cleaner),
    work: s(COLS.work), incident: s(COLS.incident),
    hasPhoto: !!photo
  };
  if (!light && photo) fields.photo = photo;
  return {
    room: room, floor: s(COLS.floor), fields: fields,
    updated_at: sd(COLS.updatedAt, true), author: s(COLS.author),
    seq: Number(s(COLS.seq) || 0), light: light
  };
}

/* ============================================================
   書き込み。前の値(prev)と食い違っていれば書かずに競合として返す（上書きしない）。
   競合はこの応答の中だけで返し、シートには記録しません（端末側で表示・保存）。
   ============================================================ */
function applyRows_(rowsRaw, author) {
  let rows = rowsRaw;
  if (typeof rows === 'string') {
    try { rows = rows ? JSON.parse(rows) : []; } catch (err) { throw new Error('rows を解釈できません'); }
  }
  if (!Array.isArray(rows)) throw new Error('rows が配列ではありません');
  const lock = LockService.getScriptLock();
  let result = { ok: true, applied: [], conflicts: [], count: 0, ignored: 0, photoSkipped: 0 };
  try {
    lock.waitLock(20000);
    const applied = [], conflicts = [];
    /* room_state 時代の key（staff:list など）が来ても、シートは作らず読み飛ばす */
    const roomRows = [];
    rows.forEach(function (r) {
      if (!r || !r.key) return;
      if (!/^(room|work|incident):/.test(String(r.key))) { result.ignored++; return; }
      roomRows.push(r);
    });

    if (roomRows.length) {
      const sh = sheet_();
      prune_(sh);                                   /* 重複行があっても必ず1本にまとめてから書く */
      const last = sh.getLastRow();
      const values = last > 1 ? sh.getRange(2, 1, last - 1, HEADERS.length).getValues() : [];
      const at = {};
      let maxSeq = 0;
      for (let i = 0; i < values.length; i++) {
        const room = String(values[i][COLS.room - 1] || '').trim();
        if (!room) continue;
        if (!at[room]) at[room] = { row: i + 2, vals: values[i] };
        const sq = Number(values[i][COLS.seq - 1] || 0);
        if (sq > maxSeq) maxSeq = sq;
      }
      const now = new Date();
      const nowStr = Utilities.formatDate(now, Session.getScriptTimeZone() || 'Asia/Tokyo', 'yyyy/MM/dd HH:mm:ss');

      roomRows.forEach(function (r) {
        const m = /^(room|work|incident):(.+)$/.exec(String(r.key));
        if (!m) return;
        const kind = m[1], room = String(m[2]).trim();
        const fields = r.fields || {};
        const prev = r.prev || {};
        let photoTooBig = false;
        const cur = at[room] ? rowToObj(at[room].vals, room, false) : null;

        /* 競合判定：送ろうとした列が、前回知っていた値と違っていれば他の端末が先に更新している */
        if (cur) {
          for (let k = 0; k < INFO_KEYS.length; k++) {
            const name = INFO_KEYS[k];
            if (!(name in fields)) continue;
            if (!(name in prev)) continue;
            const mine = canon_(name, prev[name]);
            const theirs = name === 'photo' ? canon_(name, cur.fields.photo) : canon_(name, cur.fields[name]);
            if (mine !== theirs) {
              conflicts.push({
                key: r.key, serverSeq: cur.seq,
                server: lightFields_(cur.fields), client: fields
              });
              return;
            }
          }
        } else if (Object.keys(prev).length) {
          conflicts.push({ key: r.key, serverSeq: 0, server: {}, client: fields });
          return;
        }

        /* 反映：送られた列だけを書き換える（他の列と写真はそのまま残る） */
        const row = at[room] ? at[room].vals.slice() : new Array(HEADERS.length).fill('');
        row[COLS.room - 1] = room;
        if (kind === 'room') {
          ['model', 'monitor', 'ac', 'camera', 'cleanedOn', 'cleaner'].forEach(function (name) {
            if (!(name in fields)) return;
            if (name === 'cleanedOn') {                 /* 清掃日は YYYY-MM-DD で書き込む */
              const iso = canon_('cleanedOn', fields[name]);
              row[COLS.cleanedOn - 1] = iso || String(fields[name] || '');
              return;
            }
            row[COLS[name] - 1] = fields[name];
          });
          if ('photo' in fields) {
            const ph = String(fields.photo || '');
            if (ph.length > BIG_PHOTO) {
              result.photoSkipped = (result.photoSkipped || 0) + 1;   /* 大きすぎる写真は保存せず、いまの写真を残す */
              photoTooBig = true;                                     /* 端末に知らせて縮小→再送させる */
            } else {
              row[COLS.photo - 1] = ph;
            }
          }
        } else if (kind === 'work') {
          row[COLS.work - 1] = ('work' in fields) ? fields.work : '';
        } else {
          row[COLS.incident - 1] = ('incident' in fields) ? fields.incident : '';
        }
        if (fields.floor) row[COLS.floor - 1] = fields.floor;
        row[COLS.updatedAt - 1] = nowStr;
        row[COLS.author - 1] = String(author || '');
        maxSeq += 1;
        row[COLS.seq - 1] = maxSeq;
        const target = at[room] ? at[room].row : (sh.getLastRow() + 1);
        sh.getRange(target, 1, 1, HEADERS.length).setValues([row]);
        at[room] = { row: target, vals: row };
        applied.push({ key: r.key, seq: maxSeq, photoSkipped: photoTooBig });
      });
    }

    result.applied = applied;
    result.conflicts = conflicts;
    result.count = applied.length;
  } catch (err) {
    result = { ok: false, error: String(err) };
  } finally {
    try { lock.releaseLock(); } catch (err) {}
  }
  return result;
}

function lightFields_(f) {
  const out = {};
  Object.keys(f).forEach(function (k) { if (k !== 'photo') out[k] = f[k]; });
  out.hasPhoto = !!f.photo;
  return out;
}
function norm_(v) { return (v === null || v === undefined) ? '' : String(v).trim(); }

function sheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) sh = ss.insertSheet(SHEET_NAME);
  const last = sh.getLastRow();
  const head = last > 0 ? sh.getRange(1, 1, 1, HEADERS.length).getValues()[0] : [];
  const okHead = String(head[0] || '') === HEADERS[0] && String(head[COLS.seq - 1] || '') === HEADERS[COLS.seq - 1];
  if (!okHead) {
    sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
    sh.setFrozenRows(1);
  }
  try {                                    /* 日付・更新日時の列は文字列として扱う（勝手に日付セルへ変換させない） */
    const n = Math.max(sh.getLastRow(), 2);
    DATE_COLS.forEach(function (c) { sh.getRange(2, c, n - 1, 1).setNumberFormat('@'); });
  } catch (err) { /* 書式設定に失敗しても本体は続行 */ }
  return sh;
}

101 一階
正常
異常の内容（一言） 必須

登録すると、この部屋は使用不可として扱われ、通常モード・作業モードでも赤く表示されます。すでに発生中の部屋では、下の「異常事態を解除する」で正常に戻せます（解除も全モード・全端末に反映）。

登録日時（自動反映） — 登録時に記録
